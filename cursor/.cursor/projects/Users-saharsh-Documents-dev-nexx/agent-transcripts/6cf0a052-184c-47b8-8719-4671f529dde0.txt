**`src/bot/commands/questionSolve.ts`**:

- Group responses by `selected_option`
- Find option with most votes
- Format: "Option X (selected by N/M models)"

4. **Error Handling**:

- Wrap all handlers in try/catch
- React "❌" on error

## Key Implementation Details

1. **Image Handling**: Use `ctx.getFile()` to get Telegram file URL, pass directly to AI vision model

2. **Parallel Solving**: Use `Promise.allSettled()` for concurrent model calls - ensures all results are collected even if one fails

3. **Search Grounding**: Gemini solver uses `useSearchGrounding: true` to access real-time web information

4. **No Aggregation**: All model responses are sent individually to the user (no consensus calculation)

5. **Flexible MCQ Options**: Schema uses `z.record()` to support any option format (A-Z, 1-10, I-IV, etc.)

6. **Reasoning Limit**: All solve schemas have `reasoning: z.string().max(500)` to keep responses concise

7. **Error Handling**:
   - Wrap all handlers in try/catch
   - React "❌" on error
   - Send user-friendly error message
   - Log full error to telemetry

```typescript
export const formatDollar = (value: number) => {
  const dollarFormatter = new Intl.NumberFormat("en-US", {
    style: "currency",
    currency: "USD",
    trailingZeroDisplay: "stripIfInteger",
    maximumSignificantDigits: 2,
  })
  return dollarFormatter.format(value)
}

export const getModelsData = async () => {
  try {
    const response = await fetch("https://models.dev/api.json")
    if (!response.ok) throw new Error(`HTTP ${response.status}`)
    return await response.json()
  } catch (error) {
    logger.error("Failed to fetch models data:", error)
    return null
  }
}

export const tokensUsed = async (
  usage: LanguageModelV2Usage,
  model: LanguageModelV2,
): Promise<string> => {
  const inTokens = usage.inputTokens
  const outTokens = usage.outputTokens

  try {
    const findModelCost = (
      modelProvider: string,
      modelId: string,
      modelsData: Record<string, { models: Record<string, { cost: { input: number; output: number } }> }>,
    ) => {
      modelProvider = modelProvider.split(".")[0]
      return modelsData[modelProvider].models[modelId].cost
    }

    if (!inTokens || !outTokens) throw new Error("In/Out tokens not defined")

    const modelsData = await getModelsData()
    if (!modelsData) throw new Error("Could not fetch models data.")

    const modelCost = findModelCost(model.provider, model.modelId, modelsData)
    const inCost = modelCost.input / 1e6
    const outCost = modelCost.output / 1e6
    const cost = formatDollar(inCost * inTokens + outCost * outTokens)

    return `${inTokens} -> ${outTokens} tokens (${cost})`
  } catch (error) {
    logger.warn("Could not calculate token usage:", error)
    return `${inTokens} -> ${outTokens} tokens (unknown)`
  }
}

export async function withAIOperation<T>({
  operationName,
  logInfo,
  model,
  operation,
}: {
  operationName: string
  logInfo?: string | Record<string, unknown>
  model: LanguageModelV2
  operation: () => Promise<{ usage: LanguageModelV2Usage } & T>
}): Promise<T> {
  try {
    logger.info(`${operationName}${logInfo ? ":" : ""}`, logInfo)

    const startTime = performance.now()
    const result = await operation()
    const endTime = performance.now()

    const durationSeconds = new Intl.NumberFormat("en-US", {
      trailingZeroDisplay: "stripIfInteger",
      maximumSignificantDigits: 3,
    }).format((endTime - startTime) / 1000)

    const tokensSummary = await tokensUsed(result.usage, model)

    logger.success(`${operationName} completed (${durationSeconds}s):`, tokensSummary)
    return result
  } catch (error) {
    logger.error(`Error ${operationName.toLowerCase()}:`, error)
    throw error
  }
}

export { google }
```

```typescript
import { generateObject } from "ai"
import { mcqSolveSchema, questionSolveSchema, type McqTranscription, type McqSolveResult, type QuestionTranscription, type QuestionSolveResult } from "@/core/types"
import { google, MODELS, withAIOperation } from "./index"

interface SolverResult<T> {
  name: string
  result: T | null
  error?: string
}

export async function solveMcq(mcq: McqTranscription): Promise<SolverResult<McqSolveResult>[]> {
  const prompt = formatMcqPrompt(mcq)

  const results = await Promise.allSettled(
    MODELS.solvers.map(async ({ name, model }) => {
      const result = await withAIOperation({
        operationName: `Solving MCQ (${name})`,
        model,
        operation: async () =>
          await generateObject({
            model,
            schema: mcqSolveSchema,
            tools: name.includes("Gemini") ? { google_search: google.tools.googleSearch({}) } : undefined,
            prompt,
          }),
      })
      return { name, result: result.object }
    }),
  )

  return results.map((r, i) => {
    if (r.status === "fulfilled") return r.value
    return { name: MODELS.solvers[i].name, result: null, error: String(r.reason) }
  })
}

export async function solveQuestion(question: QuestionTranscription): Promise<SolverResult<QuestionSolveResult>[]> {
  const prompt = formatQuestionPrompt(question)

  const results = await Promise.allSettled(
    MODELS.solvers.map(async ({ name, model }) => {
      const result = await withAIOperation({
        operationName: `Solving question (${name})`,
        model,
        operation: async () =>
          await generateObject({
            model,
            schema: questionSolveSchema,
            tools: name.includes("Gemini") ? { google_search: google.tools.googleSearch({}) } : undefined,
            prompt,
          }),
      })
      return { name, result: result.object }
    }),
  )

  return results.map((r, i) => {
    if (r.status === "fulfilled") return r.value
    return { name: MODELS.solvers[i].name, result: null, error: String(r.reason) }
  })
}

function formatMcqPrompt(mcq: McqTranscription): string {
  const optionsText = Object.entries(mcq.options)
    .map(([label, text]) => `${label}: ${text}`)
    .join("\n")
  return `Solve this multiple choice question:\n\n${mcq.question}\n\nOptions:\n${optionsText}${mcq.context ? `\n\nContext: ${mcq.context}` : ""}`
}

function formatQuestionPrompt(q: QuestionTranscription): string {
  return `Solve this question:\n\n${q.question}${q.context ? `\n\nContext: ${q.context}` : ""}`
}
```

```typescript
export const registerMcqTranscribeHandler = (bot: Bot) => {
  bot.command("mcq_transcribe", async (ctx) => {
    const userId = ctx.from?.id
    if (!userId) return

    if (sessions.get(userId) !== "IDLE" && sessions.has(userId)) {
      await ctx.reply("Session already active. Use /stop first.")
      return
    }

    await ctx.reply("Send an MCQ image to transcribe.")
  })

  bot.on(":photo", async (ctx) => {
    const userId = ctx.from?.id
    if (!userId || sessions.get(userId) !== "MCQ_TRANSCRIBE") return

    logger.group("MCQ Transcribe")
    try {
      await ctx.react("👀")

      const file = await ctx.getFile()
      const imageUrl = `https://api.telegram.org/file/bot${config.BOT_TOKEN}/${file.file_path}`

      const mcq = await transcribeMcq(imageUrl)

      const optionsText = Object.entries(mcq.options)
        .map(([label, text]) => `*${label}*: ${text}`)
        .join("\n")

      await ctx.reply(`*Question:*\n${mcq.question}\n\n*Options:*\n${optionsText}`, {
        parse_mode: "Markdown",
        reply_to_message_id: ctx.message?.message_id,
      })

      await ctx.react("✅")
      sessions.set(userId, "IDLE")
      logger.success("MCQ transcription complete")
    } catch (error) {
      logger.error("MCQ transcription failed:", error)
      await ctx.react("❌")
      await ctx.reply("Error transcribing image. Please try again.")
      sessions.set(userId, "IDLE")
    } finally {
      logger.groupEnd()
    }
  })
```
